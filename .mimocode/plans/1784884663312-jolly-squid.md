# KYC/Screening Cross-Layer Security Hardening Plan

## Problem Statement

KYC and sanctions/screening status gate actions at three enforcement points (contract, backend, frontend) that were built independently and drift apart. A path that skips one point reaches the others unguarded. Screening revocation in PostgreSQL has no write-back path to the on-chain `user_profile`.

**Invariant to enforce:** No state-mutating entrypoint completes for a party that is not `KycStatus::Verified` AND `ScreeningStatus::Clear`, and every backend status transition (including revocation to Blocked/Flagged) reaches `user_profile` before the next mutating action is authorized.

```rust
pub fn assert_cleared(env: &Env, party: &Address) -> Result<(), UserProfileError> {
    let key = DataKey::Profile(party.clone());
    let profile: UserProfile = env.storage().persistent().get(&key)
        .ok_or(UserProfileError::ProfileNotFound)?;
    if profile.kyc_status != KycStatus::Verified {
        return Err(UserProfileError::KycNotVerified);
    }
    if profile.screening_status != ScreeningStatus::Clear {
        return Err(UserProfileError::ScreeningNotClear);
    }
    Ok(())
}
```

## Phase 2: Backend Guard, Status Service, Propagation (14-20h)

### 2A. `KycStatusService` — single source of truth

**File:** `backend/src/modules/kyc/kyc-status.service.ts` (NEW)

```typescript
@Injectable()
export class KycStatusService {
  constructor(
    @InjectRepository(User)
    private readonly userRepo: Repository<User>,
    private readonly auditService: AuditService,
  ) {}

  async getStatus(userId: string): Promise<{ kyc: KycStatus; screening: ScreeningStatus }> {
    const user = await this.userRepo.findOneOrFail({ where: { id: userId } });
    return { kyc: user.kycStatus, screening: user.screeningStatus };
  }

  async isCleared(userId: string): Promise<boolean> {
    const { kyc, screening } = await this.getStatus(userId);
    return kyc === KycStatus.VERIFIED && screening === ScreeningStatus.CLEAR;
  }

  async setKycStatus(userId: string, status: KycStatus, actorId: string): Promise<void> {
    await this.userRepo.update(userId, { kycStatus: status });
    await this.auditService.log({ ... AuditAction.KYC_STATUS_CHANGED ... });
    await this.propagationService.enqueue(userId, status, ...);
  }

  async setScreeningStatus(userId: string, status: ScreeningStatus, actorId: string): Promise<void> {
    await this.userRepo.update(userId, { screeningStatus: status });
    await this.auditService.log({ ... AuditAction.SCREENING_STATUS_CHANGED ... });
    await this.propagationService.enqueue(userId, ..., status);
  }
}
```

### 2B. `KycEnforcementGuard`

**File:** `backend/src/modules/kyc/guards/kyc-enforcement.guard.ts` (NEW)

```typescript
@Injectable()
export class KycEnforcementGuard implements CanActivate {
  constructor(
    private readonly kycStatusService: KycStatusService,
    private readonly auditService: AuditService,
  ) {}

  async canActivate(context: ExecutionContext): Promise<boolean> {
    const request = context.switchToHttp().getRequest();
    const user = request.user;
    if (!user) throw new ForbiddenException('UNAUTHORIZED');

    const cleared = await this.kycStatusService.isCleared(user.id);
    if (!cleared) {
      await this.auditService.log({
        action: AuditAction.KYC_ENFORCEMENT_DENIED,
        entityType: 'User',
        entityId: user.id,
        performedBy: user.id,
        status: AuditStatus.FAILURE,
        level: AuditLevel.SECURITY,
        metadata: { reason: 'KYC_NOT_CLEARED' },
      });
      throw new ForbiddenException('KYC_NOT_CLEARED');
    }
    return true;
  }
}
```

### 2D. `kyc-propagation` Bull queue

**File:** `backend/src/modules/kyc/kyc-propagation.service.ts` (NEW)

```typescript
@Injectable()
export class KycPropagationService {
  constructor(
    @InjectRepository(KycPropagationOutbox)
    private readonly outboxRepo: Repository<KycPropagationOutbox>,
    private readonly blockchainService: BlockchainService,
  ) {}

  async enqueue(userId: string, walletAddress: string, kycStatus: KycStatus, screeningStatus: ScreeningStatus): Promise<void> {
    const record = this.outboxRepo.create({ userId, walletAddress, kycStatus, screeningStatus, status: 'PENDING' });
    await this.outboxRepo.save(record);
  }

  async processPending(): Promise<void> {
    const pending = await this.outboxRepo.find({ where: { status: 'PENDING' }, order: { createdAt: 'ASC' }, take: 50 });
    for (const record of pending) {
      try {
        // Submit to user_profile contract via Stellar SDK
        await this.blockchainService.updateUserProfileStatus(record.walletAddress, record.kycStatus, record.screeningStatus);
        record.status = 'SENT';
      } catch (error) {
        record.attempts += 1;
        record.lastError = error.message;
        if (record.attempts >= 5) record.status = 'FAILED';
      }
      await this.outboxRepo.save(record);
    }
  }
}
```
